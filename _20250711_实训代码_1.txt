# 增强的数据增强策略
class AdvancedTransforms:
    @staticmethod
    def get_train_transforms():
        return T.Compose([
            T.Resize((config.img_size, config.img_size)),
            T.ColorJitter(brightness=0.4, contrast=0.4, saturation=0.4, hue=0.1),
            T.RandomGrayscale(p=0.2),
            T.RandomAffine(10, translate=(0.15, 0.15), scale=(0.8, 1.2), shear=10),
            T.RandomHorizontalFlip(p=0.5),
            T.GaussianBlur(kernel_size=3, sigma=(0.1, 2.0)),
            T.ToTensor(),  # 先转换为张量
            T.RandomErasing(p=0.3, scale=(0.02, 0.1), ratio=(0.3, 3.3)),  # 在张量上操作
            T.Normalize(mean=IMAGENET_DEFAULT_MEAN, std=IMAGENET_DEFAULT_STD)
        ])
    
    @staticmethod
    def get_val_transforms():
        return T.Compose([
            T.Resize((config.img_size, config.img_size)),
            T.ToTensor(),
            T.Normalize(mean=IMAGENET_DEFAULT_MEAN, std=IMAGENET_DEFAULT_STD)
        ])

# 数据集类 - 增强版
class EnhancedDigitsDataset(Dataset):
    def __init__(self, mode='train', aug=True, transform=None):
        super(EnhancedDigitsDataset, self).__init__()
        self.mode = mode
        self.aug = aug
        self.img_size = config.img_size
        
        # 使用增强的变换
        self.transform_train = AdvancedTransforms.get_train_transforms()
        self.transform_val = AdvancedTransforms.get_val_transforms()
        
        if transform:
            self.transform = transform
        else:
            self.transform = self.transform_train if (mode == 'train' and aug) else self.transform_val

        if mode == 'test':
            self.imgs = glob.glob(data_dir['test_data'] + '*.png')
            self.labels = None
        else:
            labels = json.load(open(data_dir[f'{mode}_label'], 'r'))
            imgs = glob.glob(data_dir[f'{mode}_data'] + '*.png')
            self.imgs = [(img, labels[os.path.split(img)[-1]]) for img in imgs if os.path.split(img)[-1] in labels]
            self.digit_counts = [len(label['label']) for _, label in self.imgs]  # 记录每个样本的数字个数

    def __getitem__(self, idx):
        if self.mode != 'test':
            img, label = self.imgs[idx]
        else:
            img = self.imgs[idx]
            label = None
        img = Image.open(img).convert('RGB')

        # 应用变换
        img = self.transform(img)

        if self.mode != 'test':
            # 处理标签
            labels = label['label'][:4]
            padded_labels = labels + (4 - len(labels)) * [10]
            return img, t.tensor(padded_labels).long()
        else:
            return img, os.path.basename(self.imgs[idx])

    def __len__(self):
        return len(self.imgs)
    
    def get_digit_counts(self):
        return self.digit_counts

# 使用 Swin Transformer 的模型架构
class SwinTransformerModel(nn.Module):
    def __init__(self, class_num=11, pretrained=True, pretrained_path=None):
        super(SwinTransformerModel, self).__init__()
        
        # 使用 Swin Transformer 作为骨干网络
        if pretrained and pretrained_path and os.path.exists(pretrained_path):
            try:
                # 尝试直接加载本地权重
                print(f"Loading pretrained weights from {pretrained_path}")
                self.backbone = timm.create_model(
                    config.swin_model, 
                    pretrained=False,
                    num_classes=0,  # 不要分类头
                )
                # 加载本地权重
                state_dict = t.load(pretrained_path, map_location='cpu')
                # 移除不匹配的分类头权重
                state_dict = {k: v for k, v in state_dict.items() if not k.startswith('head.')}
                self.backbone.load_state_dict(state_dict, strict=False)
                print(f"Successfully loaded pretrained weights from {pretrained_path}")
            except Exception as e:
                print(f"Error loading pretrained weights: {e}")
                print("Falling back to non-pretrained model")
                self.backbone = timm.create_model(
                    config.swin_model, 
                    pretrained=False,
                    num_classes=0,  # 不要分类头
                )
        else:
            print("Training from scratch without pretrained weights")
            self.backbone = timm.create_model(
                config.swin_model, 
                pretrained=False,
                num_classes=0,  # 不要分类头
            )
        
        # 获取特征维度
        feat_dim = self.backbone.num_features
        
        # 位置注意力模块
        self.attn1 = nn.Sequential(
            nn.Linear(feat_dim, 1024),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(1024, 512),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(512, 1),
            nn.Sigmoid()
        )
        self.attn2 = nn.Sequential(
            nn.Linear(feat_dim, 1024),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(1024, 512),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(512, 1),
            nn.Sigmoid()
        )
        self.attn3 = nn.Sequential(
            nn.Linear(feat_dim, 1024),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(1024, 512),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(512, 1),
            nn.Sigmoid()
        )
        self.attn4 = nn.Sequential(
            nn.Linear(feat_dim, 1024),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(1024, 512),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(512, 1),
            nn.Sigmoid()
        )

        # 分类器 - 添加BatchNorm和Dropout
        self.fc1 = nn.Sequential(
            nn.Linear(feat_dim, 512),
            nn.BatchNorm1d(512),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(512, class_num)
        )
        self.fc2 = nn.Sequential(
            nn.Linear(feat_dim, 512),
            nn.BatchNorm1d(512),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(512, class_num)
        )
        self.fc3 = nn.Sequential(
            nn.Linear(feat_dim, 512),
            nn.BatchNorm1d(512),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(512, class_num)
        )
        self.fc4 = nn.Sequential(
            nn.Linear(feat_dim, 512),
            nn.BatchNorm1d(512),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(512, class_num)
        )

    def forward(self, x):
        # 通过 Swin Transformer 提取特征
        features = self.backbone(x)
        
        # 应用位置注意力
        a1 = self.attn1(features)
        a2 = self.attn2(features)
        a3 = self.attn3(features)
        a4 = self.attn4(features)

        # 加权特征
        feat1 = features * a1
        feat2 = features * a2
        feat3 = features * a3
        feat4 = features * a4

        # 分类
        fc1 = self.fc1(feat1)
        fc2 = self.fc2(feat2)
        fc3 = self.fc3(feat3)
        fc4 = self.fc4(feat4)

        return fc1, fc2, fc3, fc4

# Focal Loss 实现
class FocalLoss(nn.Module):
    def __init__(self, alpha=None, gamma=2.0, size_average=True):
        super(FocalLoss, self).__init__()
        self.alpha = alpha
        self.gamma = gamma
        self.size_average = size_average
        
        # 将alpha列表转换为张量
        if alpha is not None:
            self.alpha_tensor = t.tensor(alpha, dtype=t.float32)
        else:
            self.alpha_tensor = None

    def forward(self, inputs, targets):
        ce_loss = F.cross_entropy(inputs, targets, reduction='none')
        pt = t.exp(-ce_loss)
        
        # 如果提供了alpha，获取每个目标的alpha值
        if self.alpha_tensor is not None:
            # 将alpha张量移动到相同设备
            alpha = self.alpha_tensor.to(inputs.device)
            # 根据目标索引选择alpha值
            alpha = alpha[targets]
        else:
            alpha = 1.0
            
        focal_loss = alpha * (1 - pt) ** self.gamma * ce_loss

        if self.size_average:
            return focal_loss.mean()
        else:
            return focal_loss.sum()

# 改进的损失函数 - 结合Focal Loss和Label Smoothing
class EnhancedLoss(nn.Module):
    def __init__(self, alpha=None, gamma=2.0, smoothing=0.1, class_weights=None):
        super(EnhancedLoss, self).__init__()
        self.focal = FocalLoss(alpha, gamma)
        self.smoothing = smoothing
        self.class_weights = class_weights

    def forward(self, inputs, targets):
        # Focal Loss
        focal_loss = self.focal(inputs, targets)
        
        # Label Smoothing Cross Entropy
        log_probs = F.log_softmax(inputs, dim=-1)
        nll_loss = -log_probs.gather(dim=-1, index=targets.unsqueeze(1)).squeeze(1)
        
        # Apply class weights if provided
        if self.class_weights is not None:
            weights = self.class_weights[targets]
            nll_loss = nll_loss * weights.to(nll_loss.device)
        
        smoothed_loss = -log_probs.mean(dim=-1)
        loss = (1 - self.smoothing) * nll_loss + self.smoothing * smoothed_loss
        
        return focal_loss + loss.mean()

# 训练器 - 增强版
class EnhancedTrainer:
    def __init__(self, fold_idx=0, val=True, pretrained_path=None):
        self.device = t.device('cuda' if t.cuda.is_available() else 'cpu')
        self.fold_idx = fold_idx
        print(f"Using device: {self.device}")
        if t.cuda.is_available():
            print(f"GPU name: {t.cuda.get_device_name(0)}")

        # 确保检查点目录存在
        os.makedirs(config.checkpoints, exist_ok=True)
        print(f"Checkpoints will be saved to: {config.checkpoints}")

        # 创建数据集
        full_train_set = EnhancedDigitsDataset(mode='train')
        
        # 创建基于数字数量的分层K折交叉验证
        skf = StratifiedKFold(n_splits=config.num_folds, shuffle=True, random_state=42)
        digit_counts = full_train_set.get_digit_counts()
        
        # 获取当前折的训练和验证索引
        for fold, (train_idx, val_idx) in enumerate(skf.split(range(len(full_train_set)), digit_counts)):
            if fold == fold_idx:
                break
                
        # 创建加权采样器以处理类别不平衡
        # 修复除零问题
        digit_counts = np.array(digit_counts)
        unique_counts = np.unique(digit_counts)
        max_count = max(unique_counts)  # 获取实际最大位数
        
        # 创建覆盖所有位数的权重数组
        class_weights = np.zeros(max_count + 1, dtype=np.float32)  # 索引从0到max_count
        
        # 为每个位数计算权重
        for count in range(1, max_count + 1):  # 位数从1开始
            count_mask = (digit_counts == count)
            if np.any(count_mask):
                class_weights[count] = 1.0 / np.sum(count_mask)
            else:
                # 如果该位数没有样本，使用平均权重
                class_weights[count] = 1.0 / len(digit_counts)
        
        # 确保没有零权重
        class_weights = np.clip(class_weights, 0.001, 10.0)
        sample_weights = class_weights[digit_counts]
        sampler = WeightedRandomSampler(sample_weights, len(sample_weights), replacement=True)
        
        # 创建数据加载器
        self.train_loader = DataLoader(
            full_train_set,
            batch_size=config.batch_size,
            sampler=sampler,
            num_workers=config.num_workers,
            pin_memory=True,
            drop_last=True,
            persistent_workers=True
        )
        
        if val:
            # 创建验证集子集
            val_subset = t.utils.data.Subset(full_train_set, val_idx)
            self.val_loader = DataLoader(
                val_subset,
                batch_size=config.batch_size,
                num_workers=config.num_workers,
                pin_memory=True,
                drop_last=False,
                persistent_workers=True
            )
        else:
            self.val_loader = None

        # 创建模型 - 使用 Swin Transformer
        self.model = SwinTransformerModel(config.class_num, pretrained=pretrained_path is not None, pretrained_path=pretrained_path).to(self.device)
        self.scaler = GradScaler(enabled=config.amp)

        # 计算类别权重 - 对空字符给予更高权重
        class_weights = [1.0] * 10 + [2.0]  # 空字符权重为2.0
        class_weights_tensor = t.tensor(class_weights, dtype=t.float32).to(self.device)
        
        # 损失函数
        self.criterion = EnhancedLoss(
            alpha=class_weights,
            gamma=2.0,
            smoothing=config.label_smoothing,
            class_weights=class_weights_tensor
        )

        # 使用AdamW优化器
        self.optimizer = AdamW(
            self.model.parameters(), 
            lr=config.lr, 
            weight_decay=config.weight_decay
        )

        # 学习率调度器 - 使用余弦退火
        self.lr_scheduler = lr_scheduler.CosineAnnealingLR(
            self.optimizer, 
            T_max=config.epochs * len(self.train_loader),
            eta_min=1e-6
        )

        self.best_acc = 0
        self.no_improve_epochs = 0  # 用于早停
        self.global_step = 0

        # 加载预训练模型
        if config.pretrained and os.path.exists(config.pretrained):
            self.load_model(config.pretrained)
            if self.val_loader is not None:
                acc = self.eval()
            self.best_acc = acc
            print(f'Loaded model from {config.pretrained}, Eval Acc: {acc * 100:.2f}%')

    def train(self):
        print(f"Starting training for {config.epochs} epochs...")
        for epoch in range(config.start_epoch, config.epochs):
            start_time = time.time()
            train_acc, train_digit_acc = self.train_epoch(epoch)
            epoch_time = time.time() - start_time

            # 评估
            if (epoch + 1) % config.eval_interval == 0 and self.val_loader is not None:
                print('Starting evaluation...')
                val_loss, val_acc, val_digit_acc = self.eval()
                print(f"Epoch {epoch + 1}/{config.epochs} - "
                      f"Train: Acc={train_acc * 100:.2f}% Digit Acc={train_digit_acc * 100:.2f}% - "
                      f"Val: Loss={val_loss:.4f} Acc={val_acc * 100:.2f}% Digit Acc={val_digit_acc * 100:.2f}% - "
                      f"Time={epoch_time:.2f}s")

                # 更新学习率
                self.lr_scheduler.step()
                current_lr = self.optimizer.param_groups[0]['lr']
                print(f"Current learning rate: {current_lr:.2e}")

                # 保存最佳模型
                if val_acc > self.best_acc:
                    self.best_acc = val_acc
                    self.no_improve_epochs = 0
                    save_path = os.path.join(
                        config.checkpoints,
                        f'best_model_fold{self.fold_idx}_epoch{epoch+1}_acc{val_acc*100:.2f}.pth'
                    )
                    self.save_model(save_path)
                    print(f'Saved best model to {save_path}')
                else:
                    self.no_improve_epochs += 1
                    print(f"No improvement for {self.no_improve_epochs} epochs")
                    
                    # 早停检查
                    if self.no_improve_epochs >= config.early_stopping_patience:
                        print(f"Early stopping at epoch {epoch+1}")
                        break
            else:
                print(f"Epoch {epoch + 1}/{config.epochs} - "
                      f"Train: Acc={train_acc * 100:.2f}% Digit Acc={train_digit_acc * 100:.2f}% - "
                      f"Time={epoch_time:.2f}s")
                self.lr_scheduler.step()

            # 保存检查点
            if (epoch + 1) % config.checkpoint_interval == 0:
                save_path = os.path.join(
                    config.checkpoints, 
                    f'checkpoint_fold{self.fold_idx}_epoch{epoch+1}.pth'
                )
                self.save_model(save_path)
                print(f'Saved checkpoint to {save_path}')

    def train_epoch(self, epoch):
        self.model.train()
        total_loss = 0.0
        corrects = 0
        total_samples = 0
        correct_digits = 0
        total_digits = 0

        tbar = tqdm(self.train_loader, desc=f'Epoch {epoch + 1}/{config.epochs}')
        for i, (img, label) in enumerate(tbar):
            img, label = img.to(self.device), label.to(self.device)
            self.optimizer.zero_grad()

            # 混合精度训练
            with autocast(enabled=config.amp):
                preds = self.model(img)
                
                # 计算每个位置的损失
                loss1 = self.criterion(preds[0], label[:, 0])
                loss2 = self.criterion(preds[1], label[:, 1])
                loss3 = self.criterion(preds[2], label[:, 2])
                loss4 = self.criterion(preds[3], label[:, 3])
                
                # 总损失
                loss = (loss1 + loss2 + loss3 + loss4) / 4

            # 反向传播
            self.scaler.scale(loss).backward()
            
            # 梯度裁剪
            self.scaler.unscale_(self.optimizer)
            nn.utils.clip_grad_norm_(self.model.parameters(), max_norm=1.0)
            
            self.scaler.step(self.optimizer)
            self.scaler.update()
            
            # 更新学习率
            self.lr_scheduler.step()
            self.global_step += 1

            # 统计信息
            with t.no_grad():
                total_loss += loss.item() * img.size(0)

                # 计算整体准确率
                correct_mask = t.stack([
                    preds[0].argmax(1) == label[:, 0],
                    preds[1].argmax(1) == label[:, 1],
                    preds[2].argmax(1) == label[:, 2],
                    preds[3].argmax(1) == label[:, 3]
                ], dim=1)
                corrects += t.all(correct_mask, dim=1).sum().item()
                total_samples += img.size(0)

                # 计算字符级准确率
                correct_digits += (preds[0].argmax(1) == label[:, 0]).sum().item()
                correct_digits += (preds[1].argmax(1) == label[:, 1]).sum().item()
                correct_digits += (preds[2].argmax(1) == label[:, 2]).sum().item()
                correct_digits += (preds[3].argmax(1) == label[:, 3]).sum().item()
                total_digits += label.numel()

            # 更新进度条
            avg_loss = total_loss / total_samples
            acc = corrects / total_samples
            digit_acc = correct_digits / total_digits if total_digits > 0 else 0
            tbar.set_postfix(
                loss=f'{avg_loss:.4f}',
                acc=f'{acc * 100:.2f}%',
                digit_acc=f'{digit_acc * 100:.2f}%'
            )

        return acc, digit_acc

    def eval(self):
        self.model.eval()
        total_loss = 0.0
        corrects = 0
        total_samples = 0
        correct_digits = 0
        total_digits = 0

        with t.no_grad():
            tbar = tqdm(self.val_loader, desc='Validation')
            for img, label in tbar:
                img, label = img.to(self.device), label.to(self.device)

                preds = self.model(img)
                
                # 计算损失
                loss1 = self.criterion(preds[0], label[:, 0])
                loss2 = self.criterion(preds[1], label[:, 1])
                loss3 = self.criterion(preds[2], label[:, 2])
                loss4 = self.criterion(preds[3], label[:, 3])
                loss = (loss1 + loss2 + loss3 + loss4) / 4
                
                total_loss += loss.item() * img.size(0)

                # 计算准确率
                correct_mask = t.stack([
                    preds[0].argmax(1) == label[:, 0],
                    preds[1].argmax(1) == label[:, 1],
                    preds[2].argmax(1) == label[:, 2],
                    preds[3].argmax(1) == label[:, 3]
                ], dim=1)
                batch_corrects = t.all(correct_mask, dim=1).sum().item()
                corrects += batch_corrects
                total_samples += img.size(0)

                # 计算字符级准确率
                correct_digits += (preds[0].argmax(1) == label[:, 0]).sum().item()
                correct_digits += (preds[1].argmax(1) == label[:, 1]).sum().item()
                correct_digits += (preds[2].argmax(1) == label[:, 2]).sum().item()
                correct_digits += (preds[3].argmax(1) == label[:, 3]).sum().item()
                total_digits += label.numel()

                # 更新进度条
                avg_loss = total_loss / total_samples if total_samples > 0 else 0
                acc = corrects / total_samples if total_samples > 0 else 0
                digit_acc = correct_digits / total_digits if total_digits > 0 else 0
                tbar.set_postfix(
                    loss=f'{avg_loss:.4f}',
                    acc=f'{acc * 100:.2f}%',
                    digit_acc=f'{digit_acc * 100:.2f}%'
                )

        avg_loss = total_loss / total_samples if total_samples > 0 else 0
        return avg_loss, corrects / total_samples if total_samples > 0 else 0, correct_digits / total_digits if total_digits > 0 else 0

    def save_model(self, save_path):
        state = {
            'model_state_dict': self.model.state_dict(),
            'optimizer_state_dict': self.optimizer.state_dict(),
            'scaler_state_dict': self.scaler.state_dict(),
            'epoch': config.start_epoch,
            'best_acc': self.best_acc,
            'global_step': self.global_step,
            'fold_idx': self.fold_idx
        }
        t.save(state, save_path)
        print(f"Model saved to {save_path}")

    def load_model(self, load_path):
        if not os.path.exists(load_path):
            print(f"Warning: Model file {load_path} does not exist. Starting from scratch.")
            return

        state = t.load(load_path, map_location=self.device)
        self.model.load_state_dict(state['model_state_dict'])
        self.optimizer.load_state_dict(state['optimizer_state_dict'])
        self.scaler.load_state_dict(state.get('scaler_state_dict', {}))

        if 'best_acc' in state:
            self.best_acc = state['best_acc']
        if 'global_step' in state:
            self.global_step = state['global_step']
        if 'fold_idx' in state:
            self.fold_idx = state['fold_idx']

        print(f"Model loaded from {load_path}")

# 集成预测函数
def ensemble_predict(model_paths, num_aug=3):
    """使用多个模型进行集成预测"""
    device = t.device('cuda' if t.cuda.is_available() else 'cpu')
    print(f"Using device: {device} for prediction")
    
    # 加载所有模型
    models_list = []
    for path in model_paths:
        model = SwinTransformerModel(config.class_num).to(device)
        state = t.load(path, map_location=device)
        model.load_state_dict(state['model_state_dict'])
        model.eval()
        models_list.append(model)
    
    test_dataset = EnhancedDigitsDataset(mode='test', aug=False)
    test_loader = DataLoader(test_dataset, batch_size=config.batch_size, shuffle=False,
                             num_workers=config.num_workers, pin_memory=True, drop_last=False)

    results = []
    char_list = [str(i) for i in range(10)] + ['']

    # 测试时数据增强转换
    tta_transforms = transforms.Compose([
        transforms.Resize((config.img_size, config.img_size)),
        transforms.ColorJitter(brightness=0.1, contrast=0.1, saturation=0.1, hue=0.05),
        transforms.RandomHorizontalFlip(p=0.5),
        transforms.ToTensor(),
        transforms.Normalize(mean=IMAGENET_DEFAULT_MEAN, std=IMAGENET_DEFAULT_STD)
    ])

    def tta_predict(img_tensor, model):
        """测试时数据增强预测"""
        img_pil = to_pil_image(img_tensor.cpu().detach())
        preds_all = []
        for _ in range(num_aug):
            augmented_img = tta_transforms(img_pil).unsqueeze(0).to(device)
            with t.no_grad():
                pred = model(augmented_img)
                pred_list = [p.argmax(1).cpu().numpy() for p in pred]
                preds_all.append(pred_list)
        return preds_all

    with t.no_grad():
        tbar = tqdm(test_loader, desc='Predicting')
        for img, img_names in tbar:
            img = img.to(device)
            all_predictions = []
            
            # 对每个模型进行预测
            for model in models_list:
                model_preds = []
                
                # 原始预测
                orig_pred = model(img)
                orig_pred = [p.argmax(1).cpu().numpy() for p in orig_pred]
                model_preds.append(orig_pred)
                
                # TTA预测
                for i in range(img.size(0)):
                    tta_pred = tta_predict(img[i], model)
                    model_preds.extend(tta_pred)
                
                all_predictions.append(model_preds)
            
            # 集成预测
            final_preds = []
            for i in range(img.size(0)):
                pos_predictions = [[] for _ in range(4)]
                
                # 收集所有模型的预测
                for model_preds in all_predictions:
                    for preds in model_preds:
                        for pos in range(4):
                            pos_predictions[pos].append(preds[pos][i])
                
                # 投票集成
                final_digits = []
                for pos_preds in pos_predictions:
                    counts = np.bincount(pos_preds)
                    final_digits.append(np.argmax(counts) if len(counts) > 0 else 10)
                
                code = ''.join([char_list[d] for d in final_digits]).rstrip(' ')
                final_preds.append(code)
            
            for name, code in zip(img_names, final_preds):
                results.append([name, code])

    save_name = os.path.join(checkpoint_dir, 'ensemble_results.csv')
    df = pd.DataFrame(results, columns=['file_name', 'file_code'])
    df.to_csv(save_name, index=None)
    print(f'Ensemble results saved to {save_name}')
    return results
